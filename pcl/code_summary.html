Logical vectors

if_else(condition, true, false)

Code
library(dplyr)

tibble(x = c(-2, 0, 5)) |>
  mutate(sign = if_else(x >= 0, "nonnegative", "negative"))
# A tibble: 3 × 2
      x sign       
  <dbl> <chr>      
1    -2 negative   
2     0 nonnegative
3     5 nonnegative
  • Creates a new vector by choosing between two values based on a condition.
  • Is type-stable (returns consistent type across all rows).

case_when(...)

Code
library(dplyr)

tibble(score = c(95, 82, 71, 60)) |>
  mutate(letter = case_when(
    score >= 90 ~ "A",
    score >= 80 ~ "B",
    score >= 70 ~ "C",
    TRUE        ~ "D"
  ))
# A tibble: 4 × 2
  score letter
  <dbl> <chr> 
1    95 A     
2    82 B     
3    71 C     
4    60 D
  • Applies multiple conditions in order; first match is used.
  • Produces a single output column from many branching rules.

is.na(x)

Code
x <- c(1, NA, 3, NA)
is.na(x)
[1] FALSE  TRUE FALSE  TRUE
  • Returns a logical vector indicating which elements are missing.
  • Often used for filtering, counting, or conditional replacement.

Numbers

count(x, wt = w)

Code

tibble(team = c("A","A","B","B"), points = c(3, 2, 5, 1)) |>

  team  total_points
  <chr>        <dbl>

%/% and %% (integer division + remainder)

Code
library(dplyr)

time <- c(730, 945, 1205)  # HHMM format as numbers
hour <- time %/% 100
minute <- time %% 100
tibble(time, hour, minute)
# A tibble: 3 × 3
   time  hour minute
  <dbl> <dbl>  <dbl>
1   730     7     30
2   945     9     45
3  1205    12      5
  • %/% divides and drops the remainder (integer division).
  • %% returns the remainder (useful for splitting encoded numbers).

lead() / lag()

Code

3     3    15                4

  • lag() shifts values down to compare with the previous row.
  • lead() shifts values up to compare with the next row.

Factors

fct_reorder(f, x)

Code
library(dplyr)
library(forcats)
library(ggplot2)

df <- tibble(
  group = factor(c("A","A","B","B","C","C")),
  score = c(5, 6, 2, 3, 9, 8)
)

df |>
  mutate(group = fct_reorder(group, score, .fun = mean)) |>
  ggplot(aes(group, score)) +
  geom_boxplot()
  • Reorders factor levels using a numeric summary (here mean score).
  • Makes plots and tables follow a meaningful order.

fct_relevel(f, "level")

Code
library(forcats)

f <- factor(c("medium","low","high","low"))
fct_relevel(f, "low", "medium", "high")
[1] medium low    high   low   
Levels: low medium high
  • Manually sets the order of factor levels.
  • Useful when you want a specific “baseline” or display order.

fct_infreq() and fct_rev()

Code
library(forcats)

f <- factor(c("cat","dog","dog","dog","bird"))
fct_infreq(f)
[1] cat  dog  dog  dog  bird
Levels: dog bird cat
Code
fct_rev(fct_infreq(f))
[1] cat  dog  dog  dog  bird
Levels: cat bird dog
  • fct_infreq() orders levels from most frequent to least frequent.
  • fct_rev() reverses the level order (often for plotting).

fct_recode() / fct_collapse()

Code
library(forcats)

f <- factor(c("freshman","sophomore","junior","senior","senior"))

fct_recode(f, "1st" = "freshman", "2nd" = "sophomore")
[1] 1st    2nd    junior senior senior
Levels: 1st junior senior 2nd
Code
fct_collapse(
  f,
  "underclass" = c("freshman","sophomore"),
  "upperclass" = c("junior","senior")
)
[1] underclass underclass upperclass upperclass upperclass
Levels: underclass upperclass
  • fct_recode() renames levels without changing grouping.
  • fct_collapse() merges several levels into fewer categories.

  • Converts text into proper date/datetime types for math and plotting.

make_date() / make_datetime()

Code
library(lubridate)

make_date(2026, 2, 20)
[1] "2026-02-20"
Code
make_datetime(2026, 2, 20, 14, 30, 0)
[1] "2026-02-20 14:30:00 UTC"
  • Builds date/datetime from separate components.
  • Helpful when your data has year/month/day in separate columns.

floor_date() / ceiling_date()

Code
library(lubridate)

dt <- ymd_hms("2026-02-20 14:37:22")
floor_date(dt, "hour")
[1] "2026-02-20 14:00:00 UTC"
Code
ceiling_date(dt, "hour")
[1] "2026-02-20 15:00:00 UTC"
  • Rounds date-times down/up to a specified unit (hour/day/week/etc.).
  • Useful for aggregating by time bins.

with_tz() vs force_tz()

Code
library(lubridate)

dt <- ymd_hms("2026-02-20 12:00:00", tz = "UTC")

with_tz(dt, "America/Chicago")
[1] "2026-02-20 06:00:00 CST"
Code
force_tz(dt, "America/Chicago")
[1] "2026-02-20 12:00:00 CST"
  • with_tz() changes how the same instant is displayed in another time zone.
  • force_tz() keeps clock time but changes the assumed time zone (different instant).

x <- c(1, NA, 3)

na_if(x, sentinel)

Code

  • Converts “special codes” into real NA.

tidyr::complete(...)

Code
library(tidyr)
library(dplyr)

df <- tibble(
  student = c("A","A","B"),
  week = c(1, 3, 1),
  score = c(10, 12, 9)
)

df |>
  complete(student, week = 1:3)
# A tibble: 6 × 3
  student  week score
  <chr>   <dbl> <dbl>
1 A           1    10
2 A           2    NA
3 A           3    12
4 B           1     9
5 B           2    NA
6 B           3    NA
  • Creates rows for missing combinations of keys (implicit missing → explicit).
  • Helps you see gaps and then fill/plot time series cleanly.

anti_join(x, y)

Code
library(dplyr)

x <- tibble(id = c(1,2,3), name = c("a","b","c"))
y <- tibble(id = c(2,3))

anti_join(x, y, by = "id")
# A tibble: 1 × 2
     id name 
  <dbl> <chr>
1     1 a
  • Returns rows in x whose keys do not appear in y.
  • Useful for finding mismatches or missing records.

Joins

left_join(x, y, by = ...)

Code
library(dplyr)

students <- tibble(id = c(1,2,3), name = c("A","B","C"))
grades   <- tibble(id = c(1,3), grade = c(90, 85))

left_join(students, grades, by = "id")
# A tibble: 3 × 3
     id name  grade
  <dbl> <chr> <dbl>
1     1 A        90
2     2 B        NA
3     3 C        85
  • Keeps all rows from x and adds matching columns from y.
  • Unmatched rows get NA for new columns.

inner_join() / full_join() / right_join()

Code
library(dplyr)

inner_join(students, grades, by = "id")
# A tibble: 2 × 3
     id name  grade
  <dbl> <chr> <dbl>
1     1 A        90
2     3 C        85
Code
full_join(students, grades, by = "id")
# A tibble: 3 × 3
     id name  grade
  <dbl> <chr> <dbl>
1     1 A        90
2     2 B        NA
3     3 C        85
Code
right_join(students, grades, by = "id")
# A tibble: 2 × 3
     id name  grade
  <dbl> <chr> <dbl>
1     1 A        90
2     3 C        85
  • inner_join() keeps only rows with matches in both tables.
  • full_join() keeps everything from both; right_join() keeps all rows of y.

semi_join() / anti_join()

Code
library(dplyr)

semi_join(students, grades, by = "id")
# A tibble: 2 × 2
     id name 
  <dbl> <chr>
1     1 A    
2     3 C
Code
anti_join(students, grades, by = "id")
# A tibble: 1 × 2
     id name 
  <dbl> <chr>
1     2 B
  • semi_join() filters x to rows that have a match in y (no extra columns added).
  • anti_join() filters x to rows that do not have a match in y.

```{r}
  "upperclass" = c("junior","senior")
)
```

-   `fct_recode()` renames levels without changing grouping.
-   `fct_collapse()` merges several levels into fewer categories.


## Missing values

### `tidyr::fill(cols)`

```{r}
library(tidyr)
library(dplyr)

df <- tibble(group = c("A", NA, NA, "B", NA), value = 1:5)

df |>
  fill(group)
```

-   Fills missing values down (or up) using the nearest non-missing value.
-   Useful for “merged-cell” style data.

### `coalesce(x, 0)`

```{r}
library(dplyr)

x <- c(1, NA, 3)
coalesce(x, 0)
```

-   Replaces `NA` with the first non-missing value among arguments.
-   Great for setting defaults (e.g., missing counts become 0).

### `na_if(x, sentinel)`

```{r}
library(dplyr)

x <- c(10, -99, 7, -99)
na_if(x, -99)
```

-   Converts “special codes” into real `NA`.
-   Makes summaries and plots treat those values as missing correctly.

### `tidyr::complete(...)`

```{r}
library(tidyr)
library(dplyr)

df <- tibble(
  student = c("A","A","B"),
  week = c(1, 3, 1),
  score = c(10, 12, 9)
)

df |>
  complete(student, week = 1:3)
```

-   Creates rows for missing combinations of keys (implicit missing → explicit).
-   Helps you see gaps and then fill/plot time series cleanly.

### `anti_join(x, y)`

```{r}
library(dplyr)

x <- tibble(id = c(1,2,3), name = c("a","b","c"))
y <- tibble(id = c(2,3))

anti_join(x, y, by = "id")
```

-   Returns rows in `x` whose keys do **not** appear in `y`.
-   Useful for finding mismatches or missing records.

## Joins

### `left_join(x, y, by = ...)`

```{r}
library(dplyr)

students <- tibble(id = c(1,2,3), name = c("A","B","C"))
grades   <- tibble(id = c(1,3), grade = c(90, 85))

left_join(students, grades, by = "id")
```

-   Keeps all rows from `x` and adds matching columns from `y`.
-   Unmatched rows get `NA` for new columns.

### `inner_join()` / `full_join()` / `right_join()`

```{r}
library(dplyr)

inner_join(students, grades, by = "id")
full_join(students, grades, by = "id")
right_join(students, grades, by = "id")
```

-   `inner_join()` keeps only rows with matches in both tables.
-   `full_join()` keeps everything from both; `right_join()` keeps all rows of `y`.

### `semi_join()` / `anti_join()`

```{r}
library(dplyr)

semi_join(students, grades, by = "id")
anti_join(students, grades, by = "id")
```

-   `semi_join()` filters `x` to rows that have a match in `y` (no extra columns added).
-   `anti_join()` filters `x` to rows that do **not** have a match in `y`.
